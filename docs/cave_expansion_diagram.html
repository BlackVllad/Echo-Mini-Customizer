§2 · Bugs corregidos en esta sesión (6 en total)
🐛 Bug 1 — ADDW usa slot_idx en vez de hw_idx
# ANTES — E y D intercambiados
dispatch.append((hw_idx, new_blk * slot_idx))
# Tema E (hw=4, slot=2) → ADDW = 2×374 = 748  ✗
# Tema D (hw=3, slot=3) → ADDW = 3×374 = 1122 ✗

# DESPUÉS — usa hardware index
dispatch.append((hw_idx, new_blk * hw_idx))
# Tema E → hw=4 × 374 = 1496 ✓
# Tema D → hw=3 × 374 = 1122 ✓
🐛 Bug 2 — old_start calcula offset incorrecto de UI
# ANTES
old_start = SHARED + t_idx * (new_blk - SHARED)
# Tema B (t_idx=1): 67 + 1×307 = 374
# Pero UI de B empieza en 374+67 = 441  ✗

# DESPUÉS
old_start = t_idx * new_blk + SHARED
# Tema B: 1×374 + 67 = 441 ✓
🐛 Bug 3 — Nuevos temas usaban placeholder vacío
# ANTES — placeholder genérico sin nombre útil
new_r26.append(shared_r26[0])
new_meta.append(shared_meta[0])

# DESPUÉS — copia recurso del tema A con prefijo correcto
a_idx  = SHARED + i              # posición equivalente en tema A
new_name = f"{letter}_{orig_name}"   # ej. "F_MAINBG.BMP"
# → Import Theme puede encontrar y reemplazar el recurso
🐛 Bug 4 — _find_code_cave retornaba dirección sin alinear
# ANTES — podía retornar dirección impar o no múltiplo de 4
return cave_start          # ej. 0x13D8E (alignment=2)
# LDR PC-rel offset = 62 (62 % 4 ≠ 0) → AssertionError

# DESPUÉS — fuerza alineación a 4 bytes
aligned = (cave_start + 3) & ~3   # 0x13D8E → 0x13D90
if cave_start + cave_len - aligned >= min_size:
    return aligned   # ✓
🐛 Bug 5 — detect_patch_info() confundía la cave con el CMP original
# Problema: la cave TAMBIÉN empieza con CMP R0,#67 (0x2843)
# → detect_patch_info encontraba la cave en 0x13D90
#   antes que el B.W real en 0x3DF5A
# → devolvía is_patched=False y new_blk=441 (incorrecto)

# DESPUÉS — distingue cave por valor del primer ADDW
is_cave = (cmp_val == 0x2843 and addw_values[0] >= NEW_BLK)
is_patched = (cmp_val == 0x2800) or is_cave
🐛 Bug 6 — is_cave_dispatched() siempre devolvía False
# ANTES — buscaba B.W en la dirección devuelta por
# detect_patch_info(), que apuntaba a la cave (0x13D90)
# donde hay CMP R0,#67, no B.W → siempre False

# DESPUÉS — usa el flag is_cave directamente
def is_cave_dispatched(self):
    info = self.detect_patch_info()
    return info.get('is_cave', False)
